#include <stdlib.h>

typedef struct {
    int to;
    int cost;
} Edge;

int dfs(int node, int parent, Edge **graph, int *size)
{
    int count = 0;

    for (int i = 0; i < size[node]; i++)
    {
        int next = graph[node][i].to;
        int cost = graph[node][i].cost;

        if (next == parent)
            continue;

        count += cost;
        count += dfs(next, node, graph, size);
    }

    return count;
}

int minReorder(int n, int** connections, int connectionsSize, int* connectionsColSize)
{
    Edge **graph = calloc(n, sizeof(Edge *));
    int *size = calloc(n, sizeof(int));
    int *capacity = calloc(n, sizeof(int));

    for (int i = 0; i < connectionsSize; i++)
    {
        int a = connections[i][0];
        int b = connections[i][1];

        // Original direction: a -> b
        // If DFS goes from a to b, this edge must be reversed.
        if (size[a] == capacity[a])
        {
            capacity[a] = (capacity[a] == 0) ? 2 : capacity[a] * 2;
            graph[a] = realloc(graph[a],
                               capacity[a] * sizeof(Edge));
        }

        graph[a][size[a]].to = b;
        graph[a][size[a]].cost = 1;
        size[a]++;

        // Reverse representation: b -> a
        // Original road is b <- a, so no reversal is needed.
        if (size[b] == capacity[b])
        {
            capacity[b] = (capacity[b] == 0) ? 2 : capacity[b] * 2;
            graph[b] = realloc(graph[b],
                               capacity[b] * sizeof(Edge));
        }

        graph[b][size[b]].to = a;
        graph[b][size[b]].cost = 0;
        size[b]++;
    }

    int answer = dfs(0, -1, graph, size);

    for (int i = 0; i < n; i++)
        free(graph[i]);

    free(graph);
    free(size);
    free(capacity);

    return answer;
}
