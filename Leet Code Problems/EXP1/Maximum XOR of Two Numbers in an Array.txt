#include <stdio.h>
#include <stdbool.h>

// Helper structure for a simple hash table to handle collisions
typedef struct HashNode {
    int key;
    struct HashNode* next;
} HashNode;

#define HASH_SIZE 1021

// Insert into custom hash table
void insert(HashNode** hashTable, int key) {
    int index = abs(key) % HASH_SIZE;
    HashNode* curr = hashTable[index];
    while (curr) {
        if (curr->key == key) return;
        curr = curr->next;
    }
    HashNode* newNode = (HashNode*)malloc(sizeof(HashNode));
    newNode->key = key;
    newNode->next = hashTable[index];
    hashTable[index] = newNode;
}

// Search in custom hash table
bool search(HashNode** hashTable, int key) {
    int index = abs(key) % HASH_SIZE;
    HashNode* curr = hashTable[index];
    while (curr) {
        if (curr->key == key) return true;
        curr = curr->next;
    }
    return false;
}

// Free hash table memory
void freeTable(HashNode** hashTable) {
    for (int i = 0; i < HASH_SIZE; i++) {
        HashNode* curr = hashTable[i];
        while (curr) {
            HashNode* temp = curr;
            curr = curr->next;
            free(temp);
        }
        hashTable[i] = NULL;
    }
}

int findMaximumXOR(int* nums, int numsSize) {
    int maxResult = 0;
    int mask = 0;
    
    // Allocate hash table bucket array on heap to avoid stack overflow
    HashNode** hashTable = (HashNode**)calloc(HASH_SIZE, sizeof(HashNode*));

    // Iterate from the most significant bit (30th bit) down to 0
    for (int i = 30; i >= 0; i--) {
        // Set the i-th bit in the mask
        mask |= (1 << i);
        
        // Clear hash table for the current bit position
        freeTable(hashTable);
        
        // Store prefixes of all numbers up to the i-th bit
        for (int j = 0; j < numsSize; j++) {
            insert(hashTable, nums[j] & mask);
        }
        
        // Gready choice: target maximum if we could flip the i-th bit to 1
        int greedyTarget = maxResult | (1 << i);
        
        // Check if there are two prefixes that XOR to our greedyTarget
        for (int j = 0; j < numsSize; j++) {
            int prefix = nums[j] & mask;
            // Property: If A XOR B = Target, then A XOR Target = B
            if (search(hashTable, prefix ^ greedyTarget)) {
                maxResult = greedyTarget;
                break;
            }
        }
    }
    
    free(hashTable);
    return maxResult;
}
