#include <stdlib.h>
#include <limits.h>

int networkDelayTime(int** times, int timesSize, int* timesColSize, int n, int k)
{
    int graph[101][101];

    // Initialize graph
    for (int i = 1; i <= n; i++)
    {
        for (int j = 1; j <= n; j++)
        {
            graph[i][j] = -1;
        }
    }

    // Store edges
    for (int i = 0; i < timesSize; i++)
    {
        int u = times[i][0];
        int v = times[i][1];
        int w = times[i][2];

        graph[u][v] = w;
    }

    // Distance from k to every node
    int dist[101];
    int visited[101];

    for (int i = 1; i <= n; i++)
    {
        dist[i] = INT_MAX;
        visited[i] = 0;
    }

    dist[k] = 0;

    // Dijkstra
    for (int count = 1; count <= n; count++)
    {
        int current = -1;
        int minDist = INT_MAX;

        // Find unvisited node with smallest distance
        for (int i = 1; i <= n; i++)
        {
            if (!visited[i] && dist[i] < minDist)
            {
                minDist = dist[i];
                current = i;
            }
        }

        // No reachable node remaining
        if (current == -1)
            break;

        visited[current] = 1;

        // Update neighboring nodes
        for (int next = 1; next <= n; next++)
        {
            if (graph[current][next] != -1 && !visited[next])
            {
                int newDist = dist[current] + graph[current][next];

                if (newDist < dist[next])
                {
                    dist[next] = newDist;
                }
            }
        }
    }

    // Find maximum shortest distance
    int answer = 0;

    for (int i = 1; i <= n; i++)
    {
        if (dist[i] == INT_MAX)
            return -1;

        if (dist[i] > answer)
            answer = dist[i];
    }

    return answer;
}
